1.4 Create the Tap‑to‑Pay smart contract (CosmWasm)

Create contracts/nexa-tap/src/contract.rs:

```rust
use cosmwasm_std::{
    entry_point, to_binary, Binary, Deps, DepsMut, Env, MessageInfo, Response, StdResult, StdError,
    Addr, Uint128, Timestamp,
};
use schemars::JsonSchema;
use serde::{Deserialize, Serialize};

#[derive(Serialize, Deserialize, Clone, Debug, PartialEq, JsonSchema)]
pub struct InstantiateMsg {
    pub admin: String,
}

#[derive(Serialize, Deserialize, Clone, Debug, PartialEq, JsonSchema)]
pub enum ExecuteMsg {
    /// Merchant submits a signed tap token to claim payment
    ClaimPayment {
        user_pubkey: String,        // ephemeral public key (hex)
        amount: Uint128,
        merchant_id: String,
        nonce: u64,
        expiry: Timestamp,
        signature: Binary,          // secp256k1 signature of payload
    },
}

#[derive(Serialize, Deserialize, Clone, Debug, PartialEq, JsonSchema)]
pub enum QueryMsg {
    /// Check if nonce already used
    IsNonceUsed { user_pubkey: String, nonce: u64 },
}

#[entry_point]
pub fn instantiate(
    deps: DepsMut,
    _env: Env,
    _info: MessageInfo,
    msg: InstantiateMsg,
) -> StdResult<Response> {
    let admin = deps.api.addr_validate(&msg.admin)?;
    deps.storage.set(b"admin", &admin.as_bytes());
    Ok(Response::default())
}

#[entry_point]
pub fn execute(
    deps: DepsMut,
    env: Env,
    info: MessageInfo,
    msg: ExecuteMsg,
) -> StdResult<Response> {
    match msg {
        ExecuteMsg::ClaimPayment {
            user_pubkey,
            amount,
            merchant_id,
            nonce,
            expiry,
            signature,
        } => execute_claim(deps, env, info, user_pubkey, amount, merchant_id, nonce, expiry, signature),
    }
}

fn execute_claim(
    deps: DepsMut,
    env: Env,
    _info: MessageInfo,
    user_pubkey: String,
    amount: Uint128,
    merchant_id: String,
    nonce: u64,
    expiry: Timestamp,
    signature: Binary,
) -> StdResult<Response> {
    // 1. Check expiry
    if env.block.time > expiry {
        return Err(StdError::generic_err("Token expired"));
    }
    // 2. Check nonce not used for this user
    let nonce_key = format!("nonce_{}_{}", user_pubkey, nonce);
    if deps.storage.get(nonce_key.as_bytes()).is_some() {
        return Err(StdError::generic_err("Nonce already used"));
    }
    // 3. Recover pubkey from signature (use secp256k1)
    let payload = format!("{}:{}:{}:{}", user_pubkey, amount, merchant_id, nonce);
    let verified = deps.api.secp256k1_verify(
        payload.as_bytes(),
        &signature,
        &hex::decode(&user_pubkey).map_err(|_| StdError::generic_err("Invalid pubkey hex"))?,
    )?;
    if !verified {
        return Err(StdError::generic_err("Invalid signature"));
    }
    // 4. Transfer tokens from user to merchant
    // The user address must be derived from the pubkey (or the ephemeral key maps to a real address).
    // Here we assume user_pubkey is the base64 of the public key, and address is nexa1... derived from it.
    // For simplicity, we require the user's address to be pre-funded and passed? Better: the contract holds an allowance.
    // For now, the caller (relayer) must also send the funds separately. Simplified: user pre-approves the contract.
    // We'll use a bank send message from user to merchant.
    let user_addr = deps.api.addr_humanize(&hex::decode(&user_pubkey).unwrap())?; // placeholder; real address derivation
    let merchant_addr = deps.api.addr_validate(&merchant_id)?;
    // Transfer from user to merchant
    let msg = cosmwasm_std::CosmosMsg::Bank(cosmwasm_std::BankMsg::Send {
        to_address: merchant_addr.to_string(),
        amount: vec![cosmwasm_std::coin(amount.u128(), "unexa")],
    });
    // Mark nonce used
    deps.storage.set(nonce_key.as_bytes(), b"1");
    let res = Response::new()
        .add_message(msg)
        .add_attribute("action", "claim_payment")
        .add_attribute("user", user_addr)
        .add_attribute("merchant", merchant_addr)
        .add_attribute("amount", amount);
    Ok(res)
}
```

The contract must be deployed on the Nexa Chain with an admin. Users will pre‑approve the contract to spend their tokens (allowance pattern). For MVP, the relayer can act as a trusted party and bypass the contract by submitting a standard bank send after verifying the signature off‑chain; the contract is for full decentralisation. The algorithm provides both: an off‑chain relayer (Phase 2) that validates tokens and sends funds directly.

2.5 Tap‑to‑Pay Relayer Endpoint (src/routes/tap.ts)

Merchant POS sends the tap token here.

```typescript
router.post('/tap/claim', async (req, res) => {
  const { user_pubkey, amount, merchant_id, nonce, expiry, signature } = req.body;
  // 1. Validate expiry
  if (Date.now() > expiry) return res.status(400).json({ error: 'Expired' });
  // 2. Check nonce in DB (idempotency)
  const existing = await db('tap_transactions').where({ user_pubkey, nonce }).first();
  if (existing) return res.json({ status: 'already_processed', tx_hash: existing.tx_hash });
  // 3. Verify signature (secp256k1) using crypto library (elliptic)
  const payload = `${user_pubkey}:${amount}:${merchant_id}:${nonce}`;
  const verified = verifySignature(payload, signature, user_pubkey); // implement with secp256k1
  if (!verified) return res.status(403).json({ error: 'Invalid signature' });
  // 4. Lookup merchant settlement address
  const merchant = await db('merchants').where('id', merchant_id).first();
  if (!merchant) return res.status(404).json({ error: 'Merchant not found' });
  // 5. Send transaction on-chain
  const userAddress = pubkeyToAddress(user_pubkey); // derive nexa address from pubkey
  // For MVP, send from user's wallet directly (requires user to have funds at that address).
  // The user's wallet must have created a one-time address funded prior to tap. Simpler: the relayer pays on behalf of user if user has pre-loaded?
  // We'll assume the user's public key maps to an existing address with balance. The relayer cannot sign for user; instead the token includes a signed message that authorizes the relayer to move funds from user? 
  // Ideal: user's wallet creates a signed transaction that the relayer broadcasts. That's more secure. So the token would include a complete transaction bytes (serialized) signed by user. Then relayer just broadcasts.
  // For now, we'll implement: user's wallet generates a signed `MsgSend` and sends it to relayer, which broadcasts. So the tap token is that signed tx.
  // Let's adapt: the token contains { tx_bytes: base64 }. 
  // We'll decode and broadcast.
  const txBytes = Buffer.from(tx_bytes, 'base64');
  const result = await client.broadcastTx(txBytes);
  if (result.code !== 0) return res.status(500).json({ error: result.log });
  // 6. Store transaction
  await db('tap_transactions').insert({
    user_pubkey, amount, merchant_id, nonce, expiry, signature, tx_hash: result.transactionHash, status: 'success'
  });
  res.json({ status: 'ok', tx_hash: result.transactionHash });
});
```
